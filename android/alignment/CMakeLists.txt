cmake_minimum_required(VERSION 3.16)
project(ytm_transcription LANGUAGES C CXX)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MINGW AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND NOT CMAKE_SYSTEM_PROCESSOR)
    set(CMAKE_SYSTEM_PROCESSOR x86_64)
endif()
add_subdirectory("${WHISPER_SOURCE}" whisper EXCLUDE_FROM_ALL)
add_library(ytm_transcription SHARED transcription.cpp)
target_link_libraries(ytm_transcription PRIVATE whisper)
if(MINGW)
    target_link_options(ytm_transcription PRIVATE -static)
endif()
